# MySQL 内核 AI Agent 设计 - 基于外部大模型方案

> 版本: 1.0
> 日期: 2026-02-05
> 
> 本文档描述一种**无需自建模型训练**的 MySQL 内核 AI 增强方案，
> 通过调用外部大语言模型 (LLM) 实现智能化能力。

---

## 1. 方案对比：自训练模型 vs 外部大模型

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                       两种方案对比                                           │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  ┌────────────────────────────────────┬────────────────────────────────────┐│
│  │     自训练小模型 (原方案)          │     外部大模型 (本方案)            ││
│  ├────────────────────────────────────┼────────────────────────────────────┤│
│  │ ✗ 需要 GPU 训练资源               │ ✓ 无需训练资源                     ││
│  │ ✗ 需要收集大量训练数据            │ ✓ 无需训练数据                     ││
│  │ ✗ 需要 ML 工程师                  │ ✓ 只需 Prompt 工程                 ││
│  │ ✗ 模型需要持续维护更新            │ ✓ 模型由厂商维护                   ││

│                                                                             │
│  【结论】                                                                    │
│  外部大模型不适合毫秒级热路径决策 (如缓存替换、基数估计)                    │

│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘
```

## 4. 核心功能模块设计

### 4.1 慢查询分析与优化建议

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                   功能 #1: 慢查询分析与优化                                  │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  【触发方式】                                                                │
│  • 定时任务: 每 5 分钟分析新增慢查询                                        │
│  • 手动触发: DBA 请求分析特定 SQL                                           │
│  • 阈值触发: 执行时间 > 1s 的查询自动入队                                   │
│                                                                             │
│  【数据采集】                                                                │
│  ┌──────────────────────────────────────────────────────────────────────┐  │
│  │  -- 获取慢查询                                                        │  │
│  │  SELECT                                                               │  │
│  │      DIGEST_TEXT,                                                     │  │
│  │      COUNT_STAR as exec_count,                                        │  │
│  │      AVG_TIMER_WAIT/1e9 as avg_latency_ms,                           │  │
│  │      SUM_ROWS_EXAMINED as total_rows_examined,                        │  │
│  │      SUM_ROWS_SENT as total_rows_sent,                                │  │
│  │      FIRST_SEEN, LAST_SEEN                                            │  │
│  │  FROM performance_schema.events_statements_summary_by_digest          │  │
│  │  WHERE AVG_TIMER_WAIT > 1e9  -- 平均 > 1秒                            │  │
│  │  ORDER BY AVG_TIMER_WAIT DESC                                         │  │
│  │  LIMIT 20;                                                            │  │
```

### 5.3 LLM 调用封装

```python
# llm/client.py - 统一的 LLM 客户端

from abc import ABC, abstractmethod
from typing import Dict, Any, Optional
import hashlib
import json
import time

class LLMClient(ABC):
    """LLM 客户端抽象基类"""
    
    @abstractmethod
    def complete(self, prompt: str, **kwargs) -> str:
        pass

class OpenAIClient(LLMClient):
    """OpenAI API 客户端"""
    
    def __init__(self, api_key: str, model: str = "gpt-4-turbo"):
        from openai import OpenAI
        self.client = OpenAI(api_key=api_key)
        self.model = model
    
    def complete(self, prompt: str, system_prompt: str = None, **kwargs) -> str:
        messages = []
        if system_prompt:
            messages.append({"role": "system", "content": system_prompt})
        messages.append({"role": "user", "content": prompt})
        
        response = self.client.chat.completions.create(
            model=self.model,
            messages=messages,
            temperature=kwargs.get("temperature", 0.1),  # 低温度保证稳定输出
            max_tokens=kwargs.get("max_tokens", 4096),
            response_format={"type": "json_object"}  # 强制 JSON 输出
        )
        return response.choices[0].message.content

class AnthropicClient(LLMClient):
    """Claude API 客户端"""
    
    def __init__(self, api_key: str, model: str = "claude-3-opus-20240229"):
        import anthropic
        self.client = anthropic.Anthropic(api_key=api_key)
        self.model = model
    
    def complete(self, prompt: str, system_prompt: str = None, **kwargs) -> str:
        response = self.client.messages.create(
            model=self.model,
            max_tokens=kwargs.get("max_tokens", 4096),
            system=system_prompt or "",
            messages=[{"role": "user", "content": prompt}]
        )
        return response.content[0].text

class LLMGateway:
    """LLM 调用网关 - 统一管理限流、缓存、重试、降级"""
    
    def __init__(self, config: Dict):
        self.config = config
        self.primary_client = self._create_client(config["primary"])
        self.fallback_client = self._create_client(config.get("fallback"))
        self.cache = LLMCache(config.get("cache", {}))
        self.rate_limiter = RateLimiter(config.get("rate_limit", {}))
    
    def _create_client(self, cfg: Dict) -> Optional[LLMClient]:
        if not cfg:
            return None
        provider = cfg["provider"]
        if provider == "openai":
            return OpenAIClient(cfg["api_key"], cfg["model"])
        elif provider == "anthropic":
            return AnthropicClient(cfg["api_key"], cfg["model"])
        # ... 其他提供商
        return None
    
    def call(self, prompt: str, system_prompt: str = None, 
             use_cache: bool = True, **kwargs) -> Dict[str, Any]:
        """
        调用 LLM，返回结构化结果
        
        Returns:
            {
                "success": bool,
                "content": str,
                "cached": bool,
                "model_used": str,
                "latency_ms": int,
                "error": str (如果失败)
            }
        """
        start_time = time.time()
        
        # 1. 检查缓存
        if use_cache:
            cache_key = self._cache_key(prompt, system_prompt)
            cached = self.cache.get(cache_key)
            if cached:
                return {
                    "success": True,
                    "content": cached,
                    "cached": True,
                    "model_used": "cache",
                    "latency_ms": int((time.time() - start_time) * 1000)
                }
        
        # 2. 检查限流
        if not self.rate_limiter.allow():
            return {
                "success": False,
                "error": "Rate limit exceeded",
                "latency_ms": int((time.time() - start_time) * 1000)
            }
        
        # 3. 调用主模型
        try:
            content = self.primary_client.complete(prompt, system_prompt, **kwargs)
            result = {
                "success": True,
                "content": content,
                "cached": False,
                "model_used": self.config["primary"]["model"],
                "latency_ms": int((time.time() - start_time) * 1000)
            }
            # 缓存结果
            if use_cache:
                self.cache.set(cache_key, content)
            return result
            
        except Exception as e:
            # 4. 降级到备用模型
            if self.fallback_client:
                try:
                    content = self.fallback_client.complete(prompt, system_prompt, **kwargs)
                    return {
                        "success": True,
                        "content": content,
                        "cached": False,
                        "model_used": self.config["fallback"]["model"],
                        "latency_ms": int((time.time() - start_time) * 1000)
                    }
                except Exception as e2:
                    pass
            
            return {
                "success": False,
                "error": str(e),
                "latency_ms": int((time.time() - start_time) * 1000)
            }
    
    def _cache_key(self, prompt: str, system_prompt: str) -> str:
        content = f"{system_prompt}|||{prompt}"
        return hashlib.sha256(content.encode()).hexdigest()
```

### 5.4 慢查询分析器实现

```python
# analyzers/slow_query_analyzer.py

from typing import Dict, List, Any
import json
from jinja2 import Template

class SlowQueryAnalyzer:
    """慢查询分析器"""
    
    SYSTEM_PROMPT = """你是一个 MySQL 性能优化专家。
请分析慢查询并给出优化建议。

分析原则:
1. 优先考虑添加合适的索引
2. 考虑 SQL 改写 (避免 SELECT *, 优化 JOIN)
3. 评估改动的风险和收益
4. 给出可直接执行的 SQL 语句

请以 JSON 格式返回结果，包含以下字段:
- analysis: 问题分析
- recommendations: 优化建议列表
- confidence: 置信度 (0-1)"""

    USER_PROMPT_TEMPLATE = """分析以下慢查询:

## 查询信息
SQL: {{ sql }}
平均执行时间: {{ avg_latency_ms }}ms
执行次数: {{ exec_count }}
扫描行数: {{ rows_examined }}
返回行数: {{ rows_sent }}

## 表结构
{% for table in tables %}
### {{ table.name }} ({{ table.rows }} 行)
列: {{ table.columns | join(', ') }}
{% endfor %}

## 现有索引
{% for idx in indexes %}
- {{ idx.table }}.{{ idx.name }}: ({{ idx.columns | join(', ') }})
{% endfor %}

## EXPLAIN 输出
```
{{ explain }}
```

请给出优化建议 (JSON 格式)。"""

    def __init__(self, mysql_collector, llm_gateway):
        self.mysql = mysql_collector
        self.llm = llm_gateway
        self.prompt_template = Template(self.USER_PROMPT_TEMPLATE)
    
    def analyze_slow_queries(self, min_latency_ms: int = 1000, 
                             limit: int = 10) -> List[Dict]:
        """分析慢查询并返回优化建议"""
        
        # 1. 获取慢查询列表
        slow_queries = self.mysql.get_slow_queries(min_latency_ms, limit)
        
        results = []
        for query in slow_queries:
            result = self.analyze_single_query(query)
            results.append(result)
        
        return results
    
    def analyze_single_query(self, query: Dict) -> Dict:
        """分析单个慢查询"""
        
        # 1. 解析 SQL 获取涉及的表
        tables = self._extract_tables(query["sql"])
        
        # 2. 获取表结构
        table_info = []
        for table in tables:
            info = self.mysql.get_table_info(table)
            table_info.append(info)
        
        # 3. 获取现有索引
        indexes = self.mysql.get_indexes(tables)
        
        # 4. 执行 EXPLAIN
        explain = self.mysql.explain_query(query["sql"])
        
        # 5. 构建 Prompt
        prompt = self.prompt_template.render(
            sql=query["sql"],
            avg_latency_ms=query["avg_latency_ms"],
            exec_count=query["exec_count"],
            rows_examined=query["rows_examined"],
            rows_sent=query["rows_sent"],
            tables=table_info,
            indexes=indexes,
            explain=explain
        )
        
        # 6. 调用 LLM
        response = self.llm.call(
            prompt=prompt,
            system_prompt=self.SYSTEM_PROMPT,
            use_cache=True
        )
        
        # 7. 解析结果
        if response["success"]:
            try:
                analysis = json.loads(response["content"])
                return {
                    "query_id": query.get("digest"),
                    "sql": query["sql"],
                    "analysis": analysis,
                    "llm_model": response["model_used"],
                    "cached": response["cached"],
                    "status": "success"
                }
            except json.JSONDecodeError:
                return {
                    "query_id": query.get("digest"),
                    "sql": query["sql"],
                    "error": "Failed to parse LLM response",
                    "raw_response": response["content"],
                    "status": "parse_error"
                }
        else:
            return {
                "query_id": query.get("digest"),
                "sql": query["sql"],
                "error": response["error"],
                "status": "llm_error"
            }
    
    def _extract_tables(self, sql: str) -> List[str]:
        """从 SQL 中提取表名"""
        import sqlparse
        from sqlparse.sql import Identifier, IdentifierList
        from sqlparse.tokens import Keyword, DML
        
        tables = []
        parsed = sqlparse.parse(sql)[0]
        
        # 简化实现，实际需要更复杂的解析
        from_seen = False
        for token in parsed.tokens:
            if from_seen:
                if token.ttype is Keyword:
                    from_seen = False
                else:
                    if isinstance(token, Identifier):
                        tables.append(token.get_real_name())
                    elif isinstance(token, IdentifierList):
                        for identifier in token.get_identifiers():
                            tables.append(identifier.get_real_name())
            if token.ttype is Keyword and token.value.upper() == 'FROM':
                from_seen = True
        
        return tables
```

```python
# 智能路由: 根据任务复杂度选择模型

class SmartLLMRouter:
    """智能模型路由 - 简单任务用便宜模型"""
    
    def __init__(self, llm_gateway):
        self.llm = llm_gateway
    
    def analyze_query(self, query_info: Dict) -> Dict:
        """根据查询复杂度选择模型"""
        
        complexity = self._estimate_complexity(query_info)
        
        if complexity == "simple":
            # 简单查询: 单表，简单条件
            # 使用 GPT-3.5
            return self.llm.call(
                prompt=self._build_prompt(query_info),
                system_prompt=self.SYSTEM_PROMPT,
                model_override="gpt-3.5-turbo"
            )
        else:
            # 复杂查询: 多表 JOIN，子查询等
            # 使用 GPT-4
            return self.llm.call(
                prompt=self._build_prompt(query_info),
                system_prompt=self.SYSTEM_PROMPT,
                model_override="gpt-4-turbo"
            )
    
    def _estimate_complexity(self, query_info: Dict) -> str:
        """估计查询复杂度"""
        sql = query_info["sql"].upper()
        
        # 简单: 单表，无子查询
        if sql.count("JOIN") == 0 and sql.count("SELECT") == 1:
            return "simple"
        
        # 复杂: 多表 JOIN 或子查询
        return "complex"
```

```python
import re
from typing import Dict, Tuple

class SQLSanitizer:
    """SQL 脱敏器"""
    
    # 敏感关键词
    SENSITIVE_PATTERNS = [
        (r"'[^']*'", "'[STRING]'"),           # 字符串常量
        (r"\b\d{13,19}\b", "[CARD_NUMBER]"),   # 信用卡号
        (r"\b\d{3}-\d{2}-\d{4}\b", "[SSN]"),   # 社保号
        (r"\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b", "[EMAIL]"),
        (r"\b\d+\b", "[NUMBER]"),              # 数字 (可选)
    ]
    
    def sanitize_sql(self, sql: str, level: str = "basic") -> str:
        """
        脱敏 SQL
        
        Args:
            sql: 原始 SQL
            level: 脱敏级别
                   - basic: 只脱敏字符串常量
                   - standard: 脱敏常量 + 敏感数据
                   - paranoid: 完全脱敏 (包括表名列名)
        
        Returns:
            脱敏后的 SQL
        """
        result = sql
        
        if level in ["basic", "standard", "paranoid"]:
            # 脱敏字符串常量
            result = re.sub(r"'[^']*'", "'[STRING]'", result)
        
        if level in ["standard", "paranoid"]:
            # 脱敏敏感数据模式
            for pattern, replacement in self.SENSITIVE_PATTERNS[1:]:
                result = re.sub(pattern, replacement, result)
        
        if level == "paranoid":
            # 完全脱敏 - 替换表名和列名
            result, mapping = self._anonymize_identifiers(result)
```
